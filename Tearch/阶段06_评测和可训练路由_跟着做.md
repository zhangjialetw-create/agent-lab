## 第 20 节：训练决定“用哪个工具”的小模型

创建 `router.py`：

```python
"""用标准库训练二分类路由器：0=统计工具，1=知识检索。"""
import json
import math
from knowledge import tokens
from paths import DATA, ROOT


def read_rows(name):
    text = (DATA / name).read_text(encoding="utf-8-sig")
    return [json.loads(line) for line in text.splitlines() if line.strip()]


def vector(text, vocab):
    present = set(tokens(text))
    return [1.0] + [float(word in present) for word in vocab]


def probability(weights, values):
    z = sum(w*x for w, x in zip(weights, values))
    z = max(-40.0, min(40.0, z))
    return 1.0 / (1.0 + math.exp(-z))


def fit(rows, epochs=180):
    # 词表仅从训练集产生，不读取测试集。
    vocab = sorted({word for row in rows for word in tokens(row["question"])})
    weights = [0.0] * (len(vocab) + 1)
    examples = [(vector(row["question"], vocab),
                 int(row["tool"] == "search_notes")) for row in rows]
    losses = []
    for epoch in range(epochs):
        for values, target in examples:
            predicted = probability(weights, values)
            error = predicted - target
            for i in range(len(weights)):
                weights[i] -= 0.15 * error * values[i]
        loss = 0.0
        for values, target in examples:
            p = probability(weights, values)
            loss -= target*math.log(max(p, 1e-9))
            loss -= (1-target)*math.log(max(1-p, 1e-9))
        losses.append(loss / len(examples))
    return {"vocab": vocab, "weights": weights, "loss": [losses[0], losses[-1]]}


def predict(model, question):
    p = probability(model["weights"], vector(question, model["vocab"]))
    # 零权重时 p=0.5；固定选择统计，形成可复现的初始基线。
    tool = "search_notes" if p > 0.5 else "study_stats"
    return tool, p


def train():
    rows = read_rows("router_train.jsonl")
    model = fit(rows)
    folder = ROOT / "artifacts"
    folder.mkdir(exist_ok=True)
    path = folder / "router.json"
    path.write_text(json.dumps(model, ensure_ascii=False, indent=2), encoding="utf-8")
    print("训练损失：", [round(x, 4) for x in model["loss"]])
    print("保存：", path.name)
    return model


def load():
    return json.loads((ROOT / "artifacts" / "router.json").read_text(encoding="utf-8"))


if __name__ == "__main__":
    train()
```

运行：

```powershell
.\.venv\Scripts\python.exe router.py
```

预期出现训练损失的首末数值，并保存 `artifacts/router.json`。随附数据实测约为 `[0.3634, 0.0033]`。打开文件能看到 vocab、weights、loss。

先用一个小数字例子理解训练，再读代码：

假设模型只有一个特征“问题是否包含环境”。出现则 x=1，不出现为 0。初始权重 w=0，模型对“要查资料”的概率 p=0.5。正确标签是 y=1，误差 p-y=-0.5。如果学习率 0.15，则新权重变为 `0 - 0.15 × (-0.5) × 1 = 0.075`。下次看到这个特征，模型会更倾向知识检索。

代码只是把这个过程扩展到很多字符片段：

| 函数/语句 | 含义 |
|---|---|
| read_rows | 逐行读取带标签的数据 |
| vocab | 只从训练问题提取字符片段作为词表 |
| vector | 出现过的片段记为 1，否则 0；开头的常数 1 对应偏置 |
| sum(w*x ...) | 特征乘权重再求和，得到一个分数 z |
| sigmoid | 把分数映射到 0—1，表示预测倾向 |
| target | 正确标签；知识检索为 1，统计为 0 |
| predicted-target | 本轮预测与目标之间的偏差 |
| weights[i] -= ... | 根据偏差调整每个参数 |
| epoch | 训练数据完整学一遍 |
| loss | 衡量模型给正确标签的概率是否足够高 |

这是逻辑回归二分类器，确实通过数据更新参数，但不是大语言模型。它能作为 Harness 的路由组件使用。

不要只看训练损失下降；那只说明更适应训练样本，还要看保留题。

## 第 21 节：训练前后用相同题目比较

创建 `evaluate.py`：

```python
import json
import argparse
from paths import ROOT
from harness import run
import router


def evaluate_router():
    rows = router.read_rows("router_test.jsonl")
    model = router.load()
    baseline = {**model, "weights": [0.0] * len(model["weights"])}
    for name, candidate in [("before", baseline), ("after", model)]:
        correct = 0
        for row in rows:
            prediction, probability = router.predict(candidate, row["question"])
            correct += prediction == row["tool"]
            print(name, row["question"], prediction, round(probability, 3))
        print(name, f"路由准确率：{correct}/{len(rows)}")


def evaluate_harness(backend):
    rows = [
        {"question": "统计学习总分钟", "contains": "75", "source": "sessions.json"},
        {"question": "如何创建虚拟环境", "contains": "venv", "source": "python-env.md#0"},
        {"question": "Git 如何保存版本", "contains": "commit", "source": "git.md#0"},
        {"question": "量子海豚XYZ", "contains": "未找到", "source": None},
    ]
    results = []
    for row in rows:
        result = run(row["question"], backend)
        passed = result["status"] == "ok" and row["contains"] in result["answer"]
        if row["source"]:
            passed = passed and row["source"] in result["citations"]
        else:
            passed = passed and result["citations"] == []
        results.append({"question": row["question"], "passed": passed, **result})
    folder = ROOT / "artifacts"
    folder.mkdir(exist_ok=True)
    path = folder / f"eval-{backend}.json"
    path.write_text(json.dumps(results, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"教学冒烟评测：{sum(r['passed'] for r in results)}/{len(results)}")
    print("报告：", path.name)


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--router", action="store_true")
    parser.add_argument("--backend", choices=["mock", "trained", "ollama"], default="mock")
    args = parser.parse_args()
    if args.router:
        evaluate_router()
    else:
        evaluate_harness(args.backend)
```

动手改进实验：

1. 先把当前测试报告保存，记下日期、模型和代码提交。
2. 自己写 20 个问题，其中 10 个统计、10 个知识检索。
3. 其中 10 个只用于开发，另 10 个封存作最终保留题；不要从保留题复制措辞补训练。
4. 把开发失败归为“选错工具、没检索到、答案不忠实、格式不合法、超时”。
5. 每次只改一项：训练样本，或检索，或 Skill，或提示词。
6. 复跑开发集；确定后只跑一次最终保留题，报告真实成功率、平均耗时和工具次数。

改了知识库不需要重训路由器；改了两类任务定义或新增工具类别，则需要重新设计标签和模型。当前训练器只支持两种工具。

## 第 22 节：测试与评测分工不同

新建 tests 文件夹及 `tests/test_core.py`：

```python
import json
import tempfile
import unittest
from pathlib import Path
from harness import run, validate
from knowledge import search
from tools import execute, study_stats
from router import fit, predict, read_rows
from skills import load_skill


class CoreTests(unittest.TestCase):
    def test_stats(self):
        self.assertEqual(study_stats()["total"], 75)

    def test_retrieval(self):
        self.assertEqual(search("如何创建虚拟环境")[0]["source"], "python-env.md#0")

    def test_no_match(self):
        self.assertEqual(search("量子海豚XYZ"), [])

    def test_tool_allowlist(self):
        with self.assertRaises(ValueError):
            execute("powershell", "Remove-Item")

    def test_skill_allowlist(self):
        with self.assertRaises(ValueError):
            load_skill("../../outside")

    def test_bad_action(self):
        with self.assertRaises(ValueError):
            validate({"action": "delete", "query": "", "answer": "", "citations": []})

    def test_wrong_field_type(self):
        with self.assertRaises(ValueError):
            validate({"action": "finish", "query": 1, "answer": "", "citations": []})

    def test_run_and_trace(self):
        with tempfile.TemporaryDirectory() as folder:
            result = run("总分钟", log_dir=Path(folder))
            self.assertEqual(result["status"], "ok")
            self.assertIn("75", result["answer"])
            events = [json.loads(line) for line in
                      Path(result["log"]).read_text(encoding="utf-8").splitlines()]
            self.assertEqual(events[-1]["summary"]["tool_calls"], 1)

    def test_budget(self):
        with tempfile.TemporaryDirectory() as folder:
            result = run("总分钟", max_steps=1, log_dir=Path(folder))
            self.assertEqual(result["status"], "failed")
            self.assertIn("预算", result["answer"])

    def test_fake_citation(self):
        def chooser(question, observations, *args):
            if not observations:
                return {"action": "study_stats", "query": "", "answer": "", "citations": []}, {}
            return {"action": "finish", "query": "", "answer": "75", "citations": ["fake.md"]}, {}
        with tempfile.TemporaryDirectory() as folder:
            result = run("总分钟", log_dir=Path(folder), decision_fn=chooser)
            self.assertEqual(result["status"], "failed")

    def test_training_changes_parameters(self):
        model = fit(read_rows("router_train.jsonl"), epochs=30)
        self.assertTrue(any(abs(w) > 0.01 for w in model["weights"]))
        self.assertLess(model["loss"][-1], model["loss"][0])
        self.assertEqual(predict(model, "解释虚拟环境的用途")[0], "search_notes")


if __name__ == "__main__":
    unittest.main()
```

```powershell
.\.venv\Scripts\python.exe -m unittest discover -s tests -v
```

固定随附文件下预期 11 tests，OK。`discover` 查找测试；`-s tests` 指定目录；`-v` 显示每个名称。

重点读三个测试：

- `assertEqual(study_stats()["total"], 75)`：工具数值必须符合固定样本。
- `with self.assertRaises(ValueError)`：这个输入应当被拒绝；拒绝本身就是正确行为。
- fake_citation：假决策故意编造来源，检查 Harness 能不能挡住。

测试检查程序规则；评测检查任务表现。规则测试全通过，模型仍可能答错，二者都要看。

如果你改变了知识文件或学习记录，先判断失败是否来自合理的数据变化。需要复现基准时，在单独练习副本中用随附数据复跑，不要把检查标准随意改到“全部通过”为止。

在学习记录中填写：

```text
日期：
代码提交：
backend 与模型：
数据集版本：
训练前后分数：
平均耗时 / 工具次数：
最有代表性的失败题：
此次只改变的因素：
下次需要验证的假设：
```

最后做一次 Git 提交。先 `git status` 和 `git diff`，再 `git add` 明确列出的源文件与数据；训练权重和日志仍留在本地 artifacts/runs，不加入代码提交。

完成标准：你能解释“小模型权重变了什么”“是否用过测试题训练”“这个分数能说明什么”。此时已经拥有一个可训练组件接入的 Harness。
